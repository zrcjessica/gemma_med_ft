% ---------------------------------------------------------------- setup
\begin{frame}{Setup}
  \begin{itemize}
    \item \textbf{Two arms}, identical seed and data order:
      \texttt{gemma-3-270m-it} and \texttt{gemma-3-1b-it}, full FT,
      lr $1\mathrm{e}{-5}$, completion-only loss.
    \item \textbf{Log-spaced checkpoints} $\{1,2,4,\dots,4096\}$ + final (4882).
      $t=0$ is the untouched base, prepended by the probe.
    \item \textbf{Lens refit per checkpoint} on a frozen 1000-sequence generic
      (non-medical) WikiText corpus, taken from \texttt{jlens}'s own
      \texttt{load\_wikitext\_prompts} --- the lens is a function of the weights.
    \item \textbf{Concordance} on the frozen 193-prompt probe set:
      50 \texttt{general} + 50 \texttt{medical} (hand-authored one-hop cloze,
      style-matched control pairs) + 93 \texttt{multihop} from
      \texttt{jlens}'s \texttt{lens-eval-multihop.json}.
    \item \textbf{Behavioral} trajectory in parallel: MedQA, MedMCQA, PubMedQA,
      reported as \emph{parsed} accuracy with parse rate alongside.
  \end{itemize}
  \vfill
  {\footnotesize All lens numbers below are the \textbf{mean over the last
  $n/5$ source layers} (the probe's \texttt{kl\_final} / \texttt{top1\_final}):
  3 layers at 270m, 5 at 1b. Where the single final layer disagrees, it is
  called out explicitly --- see the read-out caveat.
  Coverage: 14 points (270m), 15 points (1b) --- both complete.}
\end{frame}

% ---------------------------------------------------------------- probe step
\begin{frame}{The probe step in detail}
  \small
  Runs in \texttt{.venv-jlens} (\texttt{transformers} 5.14.1 + \texttt{jlens}
  @ \texttt{581d398}) --- a second env, because the training pin 4.53.2 is
  hard-incompatible. Imports nothing from \texttt{gemma\_med}.

  \vspace{0.15em}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Per checkpoint} (base prepended as step 0, then
      \texttt{checkpoint-*}):
      \begin{enumerate}\setlength{\itemsep}{0em}
        \item \textbf{Load} $\rightarrow$ \texttt{jlens.from\_hf(hf, tok)}.
          Forced text-only \texttt{Gemma3ForCausalLM}; any loader leaving a
          weight missing is \emph{rejected}, not accepted.
        \item \textbf{Fit} a fresh lens: \texttt{jlens.fit(model, corpus,}
          \texttt{dim\_batch=16, max\_seq\_len=128)} on the frozen 1000-seq
          WikiText corpus. \emph{Dominates runtime.}
        \item \textbf{Concordance}: for each of the 193 probes,
          \texttt{lens.apply(model, prompt,} \texttt{positions=[-1])}
          $\rightarrow$ per source layer, top-1 agreement and
          $\mathrm{KL}(\text{model}\,\|\,\text{lens})$ at the last token.
        \item \textbf{Drift} vs.\ $t{=}0$, per layer: relative Frobenius
          $\|J_l - J_l^{0}\| / \|J_l^{0}\|$ and cosine.
      \end{enumerate}
    \end{column}
    \begin{column}{0.43\textwidth}
      \textbf{Invariants that make the trajectory comparable}
      \begin{itemize}\setlength{\itemsep}{0em}
        \item \textbf{One load path for all points.} Letting \texttt{Auto*}
          pick per-path would read $t{=}0$ through the multimodal wrapper and
          later points through the text model --- two forwards on one
          trajectory, silently corrupting drift.
        \item \textbf{Lens refit every time.} $J$ is a function of the weights.
        \item \textbf{Corpus and probe set frozen} across all checkpoints and
          sizes.
        \item \textbf{$t{=}0$ lens cached in fp32}, so a resumed run's drift
          baseline matches a fresh run's exactly.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.1em}
  {\scriptsize \textbf{Output.} One \texttt{fsync}'d JSON row per checkpoint in
  \texttt{metrics.jsonl}, plus W\&B scalars keyed by training step. Resumable:
  completed steps are skipped and the fit checkpoints every 50 prompts, so a
  requeue does not wipe the trajectory.}
\end{frame}

% ---------------------------------------------------------------- headline
\begin{frame}{Four findings}
  \begin{enumerate}
    \item \textbf{Drift has a sharp onset, then saturates.} The Jacobian
      finishes moving in the first ${\sim}100$ steps; the remaining 97\% of
      training adds nothing.
    \item \textbf{Medical faithfulness improves at 1b, is flat at 270m ---
      and generic faithfulness degrades at both.} The cost of medical SFT is
      paid on the out-of-domain cues.
    \item \textbf{Behavior confirms the split.} 1b gains $+5$--$7$ pp parsed
      accuracy; 270m gains nothing and only churns format.
    \item \textbf{Drift leads format collapse, both lead accuracy.} The lens
      finishes moving before the model has learned any medicine.
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------- drift
\begin{frame}{1. Drift onsets sharply, then saturates}
  \centering
  \begin{tabular}{lccc}
    \toprule
    & onset & plateau & final relfro / cos (layer mean) \\
    \midrule
    \texttt{270m-it} & steps 32--64 & ${\sim}$step 128 & 0.249 / 0.976 \\
    \texttt{1b-it}   & steps 8--32  & ${\sim}$step 64  & 0.279 / 0.956 \\
    \bottomrule
  \end{tabular}

  \vspace{1.2em}
  \begin{itemize}
    \item Nothing moves for the first ${\sim}8$ steps
      (mean rel.\ Frobenius $\leq 0.004$).
    \item Then the Jacobian swings hard over roughly \emph{one decade} of
      steps, and flatlines.
    \item 1b: $0.299$ at step 64 $\rightarrow$ $0.279$ at step 4882.
      The last 4000 steps buy \textbf{zero} additional lens movement.
  \end{itemize}
  \vfill
  {\footnotesize Most robust result --- holds at both sizes, and under either
  the layer mean or the per-layer max (max: 0.349 / 0.398).}
\end{frame}

\begin{frame}{2. Medical holds or improves; generic degrades}
  \small
  Lens$\leftrightarrow$model KL, mean over the last $n/5$ layers
  (the probe's \texttt{kl\_final}) --- \emph{lower = lens predicts the model
  better}. Base $\rightarrow$ final checkpoint.

  \vspace{0.7em}
  \centering
  \begin{tabular}{lccc}
    \toprule
    & KL \texttt{general} & KL \texttt{medical} & KL \texttt{multihop} (OOD) \\
    \midrule
    \texttt{270m-it} & $3.407 \rightarrow 3.497$ \; (\bad{$+0.09$})
                     & $5.388 \rightarrow 5.340$ \; (\good{$-0.05$})
                     & $1.958 \rightarrow 1.975$ \; (\bad{$+0.02$}) \\
    \texttt{1b-it}   & $2.684 \rightarrow 3.229$ \; (\bad{$+0.54$})
                     & $4.682 \rightarrow 4.314$ \; (\good{$-0.37$})
                     & $2.423 \rightarrow 3.430$ \; (\bad{$+1.01$}) \\
    \midrule
    \multicolumn{4}{l}{\footnotesize top-1 \texttt{medical}: \;
      270m $0.260 \rightarrow 0.260$ (flat) \quad\quad
      1b $0.448 \rightarrow$ \good{$0.572$}} \\
    \bottomrule
  \end{tabular}

  \vspace{1.0em}
  \begin{itemize}
    \item \textbf{Same direction at both sizes}, differing in magnitude:
      medical flat-to-better, \texttt{general} and \texttt{multihop} worse.
    \item \textbf{1b moves $6$--$59\times$ further} on every cue
      (\texttt{general} $6.0\times$, \texttt{medical} $7.6\times$,
      \texttt{multihop} $59\times$). At 270m the lens barely registers the
      fine-tune --- consistent with 270m learning no medicine behaviorally.
  \end{itemize}
  \vfill
  {\footnotesize \bad{Read-out caveat:} at the \emph{single} final layer, 270m
  medical reads $1.861 \rightarrow 2.137$ ($+0.28$, \emph{worse}) --- the one
  sign flip across all six cue/size pairs. Do not read 270m medical as a
  degradation: under the layer mean used here it is flat on both KL ($-0.05$)
  and top-1 ($0.260 \rightarrow 0.260$). At the single final layer its top-1
  \emph{also} falls ($0.44 \rightarrow 0.38$), so the two read-outs disagree on
  that cue in both metrics.}
\end{frame}

\begin{frame}{2. What is really happening: the domain gap narrows}
  \small
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Within each arm the \texttt{medical} cue starts ${\sim}2$ nats less
      faithful than \texttt{general}, and both arms narrow that gap --- but 1b
      narrows it \textbf{6.6$\times$ harder}.
      \vspace{0.5em}
      \begin{tabular}{lccc}
        \toprule
        & gap $t{=}0$ & gap final & $\Delta$ \\
        \midrule
        \texttt{270m-it} & 1.981 & 1.843 & $-0.138$ \\
        \texttt{1b-it}   & 1.998 & 1.085 & $\mathbf{-0.913}$ \\
        \bottomrule
      \end{tabular}
      \vspace{0.6em}

      Crucially the gap closes \textbf{from both ends} --- medical improves
      \emph{and} general regresses toward it. It is not a free gain.
      \vspace{0.4em}

      \texttt{multihop}, the held-out externally-sourced cue, is the cleanest
      read and degrades at \emph{both} sizes ($+0.02$ / $+1.01$).
      \vspace{0.4em}

      \textbf{Generic-domain faithfulness is what medical SFT costs you.}
      \vspace{0.4em}

      {\footnotesize The $t{=}0$ gaps landing $0.017$ apart is a coincidence,
      not a finding --- absolute KL is not comparable across sizes. The claim
      is the within-arm $\Delta$, and the $6.6\times$ inherits that caveat.}

\begin{frame}{3. Behavior confirms the two scales differ}
  Parsed accuracy (parse rate in parentheses):

  \vspace{0.8em}
  \centering
  \begin{tabular}{lccc}
    \toprule
    & MedQA & MedMCQA & PubMedQA \\
    \midrule
    \texttt{270m-it} $t{=}0$   & 24.6 \; (69\%)  & 27.2 \; (13\%)  & 54.5 \; (7\%) \\
    \texttt{270m-it} final     & 25.5 \; (49\%)  & 25.0 \; (49\%)  & 48.0 \; (5\%) \\
    \midrule
    \texttt{1b-it} $t{=}0$     & 31.9 \; (100\%) & 32.6 \; (100\%) & 52.6 \; (100\%) \\
    \texttt{1b-it} final       & \good{\textbf{38.5}} \; (\bad{25\%})
                               & \good{\textbf{37.8}} \; (\bad{75\%})
                               & \good{\textbf{58.1}} \; (97\%) \\
    \bottomrule
  \end{tabular}

  \vspace{1.2em}
  \begin{itemize}
    \item \textbf{270m learns nothing.} MedQA/MedMCQA sit near chance
      throughout; every apparent movement is format churn (MedMCQA parse
      13\%$\rightarrow$49\%; PubMedQA swings 5\%$\rightarrow$88\%$\rightarrow$5\%).
    \item \textbf{1b gains $+6.6$ / $+5.2$ / $+5.5$ pp} parsed --- while format
      compliance collapses, worst on MedQA ($100\% \rightarrow 25\%$).
  \end{itemize}
  \vfill
  {\footnotesize Reported parsed, not raw: the raw MedQA number would show a
  \emph{loss} purely from the parse-rate collapse.}
\end{frame}

% ---------------------------------------------------------------- caveats
\begin{frame}{Caveats}
  \begin{itemize}
    \item \textbf{Top-1 is coarse.} $n{=}50$ prompts for
      \texttt{general}/\texttt{medical} ($93$ for \texttt{multihop}), so
      \emph{per-layer} top-1 quantizes at $0.02$; the reported statistic
      averages 3 layers (270m) or 5 (1b), giving a $1/150$--$1/250$ grain that
      makes small moves look more precise than they are. Single-checkpoint
      wiggles of $\pm 0.04$ are still noise. \emph{Trust the KL trends, not
      top-1.}
    \item \textbf{Read-out choice is not innocent.} Layer-mean and final-layer
      KL disagree in sign on 270m medical. Every claim here uses the layer mean;
      an earlier draft of this deck used the final layer and reported a
      270m$\leftrightarrow$1b \emph{inversion} that does not survive. Report the
      statistic, not just the number.
    \item \textbf{Two sizes is not a scaling trend.} ``1b moves far, 270m barely
      moves'' rests on exactly one pair, and 270m is small enough that it may
      simply be too weak to learn the task at all. Needs 4b.
    \item \textbf{The \texttt{general}/\texttt{medical} cues are
      hand-authored}, not sampled from a corpus --- the domain-gap result
      inherits whatever bias is in those 100 prompts. \texttt{multihop} is the
      one cue with an external, independently-constructed source.
    \item \textbf{Coverage gaps.} \texttt{4b\_it\_validate\{,2\}} are 5-point
      wrapper-validation spikes (base + steps 1--8), not trajectories.
      No \texttt{-pt} arm exists at any size --- the pt-vs-it starting-point
      axis is entirely unstarted.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- next
\begin{frame}{Next}
  \begin{enumerate}
    \item \textbf{4b-it full trajectory} --- the decisive test of whether
      ``1b moves far, 270m barely moves'' is a scale effect or a property of
      one arm, and whether the layer-mean/final-layer disagreement recurs.
    \item \textbf{Widen the \texttt{general}/\texttt{medical} cues} beyond the
      100 hand-authored prompts, to check the domain-gap collapse is not an
      artifact of how those pairs were written.
    \item \textbf{Open a \texttt{-pt} arm} at 1b: does a non-aligned starting
      point show the same drift-before-competence ordering?
    \item \textbf{Anchor against \texttt{medgemma-27b-text-it}} on the same
      probe set to place the trajectory endpoints on an absolute scale.
  \end{enumerate}
\end{frame}
