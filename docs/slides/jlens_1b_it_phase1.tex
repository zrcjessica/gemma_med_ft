\begin{frame}{The three metrics, defined}
  Let $J_\ell$ be the fitted Jacobian at source layer $\ell$, $J_\ell^{0}$ its
  $t{=}0$ (base) value, and at a probe position let $p$ be the model's next-token
  distribution and $q_\ell$ the lens read-out transported from layer $\ell$.
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \textbf{\textcolor{aqua}{Drift}} \; \footnotesize(per layer, then mean)
      \[
        r_\ell=\frac{\lVert J_\ell-J_\ell^{0}\rVert_F}{\lVert J_\ell^{0}\rVert_F},
        \quad
        c_\ell=\frac{\langle J_\ell, J_\ell^{0}\rangle}
                    {\lVert J_\ell\rVert\,\lVert J_\ell^{0}\rVert}
      \]
      \footnotesize\color{muted}
      How far the lens has moved from the base. $r$ rises from 0,
      $c$ falls from 1.
    \end{column}
    \begin{column}{0.34\textwidth}
      \textbf{\textcolor{genblue}{Faithfulness KL}}
      \[
        \mathrm{KL}(p\,\|\,q_\ell)=\sum_v p_v\log\frac{p_v}{q_{\ell,v}}
      \]
      \footnotesize\color{muted}
      Divergence between what the model says and what the lens reads out at
      layer $\ell$. Lower $=$ more faithful; sensitive at small probe sets.
    \end{column}
    \begin{column}{0.30\textwidth}
      \textbf{\textcolor{medorange}{Top-1 agreement}}
      \[
        a_\ell=\mathbf{1}\!\left[\hat\imath_\ell=\hat\imath\right]
      \]
      \footnotesize\color{muted}
      $\hat\imath_\ell=\arg\max q_\ell$, $\hat\imath=\arg\max p$: does the
      lens's top token match the model's? Coarser than KL, so noisier.
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \hrule height 0.4pt
  \vspace{0.5em}
  {\small
   \textbf{What the plots show.} Drift panels: $\overline{r},\overline{c}$ meaned
   over all source layers. KL / top-1 panels: $\overline{\mathrm{KL}}$ and
   $\overline{a}$ over the \emph{final} $\lceil n_{\text{layers}}/5\rceil$ layers
   (where the read-out has resolved), averaged over the probe prompts and split
   by \gen\ / \med\ arm. All read at the last token position.}
\end{frame}

\begin{frame}{Drift: fast reorganization, then plateau}
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \includegraphics[width=\textwidth]{\figpath{fig_drift.pdf}}
    \end{column}
    \begin{column}{0.36\textwidth}
      \small
      $J_\ell$ barely moves for the first $\sim$8 steps, reorganizes sharply
      over steps 16--64, and \textbf{plateaus at rel.\ Frobenius $\approx0.30$}
      (cosine $\approx0.95$) for the remaining $\sim$4800 steps.
      \\[0.6em]
      The representation's input$\rightarrow$output coupling settles early;
      later training is behavioral refinement, not further lens movement.
      \\[0.6em]
      \textcolor{medorange}{\textbf{Caveat:}} \texttt{checkpoint-1} is
      bit-identical to $t{=}0$ (drift exactly 0) --- saved before the first
      optimizer step landed. A wasted probe point; fix the schedule.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Findings \& next steps}
  \textbf{Findings}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item $J_\ell$ reorganizes in the first $\sim$64 steps, then plateaus ---
          the strongest, cleanest signal in the run.
    \item The lens is measurably less faithful on \med\ than \gen\ content
          ($\sim$1 nat KL), stable across the trajectory.
    \item KL is smooth and readable; top-1 is noise-limited at the current
          probe-set size.
  \end{itemize}
  \vspace{0.4em}
  \textbf{Fix before scaling}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item \texttt{checkpoint-1} $\equiv$ base --- drop or shift the first
          log-schedule point.
    \item Non-resumable fit: 12b/27b need a per-step checkpoint path before a
          long fit can survive a requeue.
  \end{itemize}
  \vspace{0.4em}
  \textbf{Next}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item Expand + re-freeze the probe set; re-probe.
    \item Pair with the behavioral trajectory (MedQA / MedMCQA / PubMedQA).
    \item Second real trajectory: 1b-\emph{pt} (the pt-vs-it starting-point axis).
  \end{itemize}
\end{frame}
